cmake_minimum_required(VERSION 3.16)
project(HSF_KT VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

add_library(hsf_core STATIC cpp/src/stat_engine.cpp)
set_target_properties(hsf_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(hsf_core PUBLIC cpp/include)
target_link_libraries(hsf_core PUBLIC Threads::Threads)
add_library(hsf_stat SHARED cpp/src/c_api.cpp)
target_compile_definitions(hsf_stat PRIVATE HSF_BUILD_DLL)
target_link_libraries(hsf_stat PRIVATE hsf_core)
add_executable(bench_and_test cpp/apps/bench_and_test.cpp)
target_link_libraries(bench_and_test PRIVATE hsf_core)

foreach(target hsf_core hsf_stat bench_and_test)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /O2)
  else()
    target_compile_options(${target} PRIVATE -O3 -Wall -Wextra -Wpedantic)
  endif()
  if(MINGW)
    target_link_options(${target} PRIVATE -static-libgcc -static-libstdc++ -static)
  endif()
endforeach()
enable_testing()
add_test(NAME native_rigor COMMAND bench_and_test --test-only)
